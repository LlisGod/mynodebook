### 题解

```sql
USE mydata;

#请在此处添加实现代码

########## Begin ##########

WITH alldates AS(

    SELECT DISTINCT date FROM login

),

first_date AS(

    SELECT user_id, MIN(date) AS first_date

    FROM login

    GROUP BY user_id

),

new_user AS(

    SELECT first_date, COUNT(user_id) as new_user

    FROM first_date

    GROUP BY first_date

),

reup_user AS(

    SELECT fd.first_date, COUNT(DISTINCT l.user_id) as reup_user

    FROM first_date fd

    LEFT JOIN login l on fd.user_id=l.user_id

    AND datediff(l.date,fd.first_date)=1

    GROUP BY fd.first_date

)

SELECT

    d.date as date,

    IFNULL(ROUND(ru.reup_user/nu.new_user,3),0.000) AS p

FROM alldates d

LEFT JOIN new_user nu on nu.first_date=d.date

LEFT JOIN reup_user ru on ru.first_date=d.date

ORDER BY d.date;

  

########## End ##########
```

## 题解
```sql


WITH filtered_orders AS (
    SELECT 
        user_id, 
        date, 
        id
    FROM order_info
    WHERE 
        date > '2021-10-15' 
        AND status = 'completed' 
        AND product_name IN ('C++', 'Java', 'Python')
),
ranked_orders AS (
    SELECT 
        user_id,
        date,
        ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY date, id) AS rn,
        COUNT(*) OVER (PARTITION BY user_id) AS cnt
    FROM filtered_orders
)
SELECT 
    user_id,
    MAX(CASE WHEN rn = 1 THEN date END) AS first_buy_date,
    MAX(CASE WHEN rn = 2 THEN date END) AS second_buy_date,
    MAX(cnt) AS cnt
FROM ranked_orders
WHERE cnt >= 2
GROUP BY user_id
ORDER BY user_id;

```

### 题解

```sql
WITH filtered_orders AS (
    SELECT 
        user_id,
        product_name,
        status
    FROM order_info
    WHERE date BETWEEN '2021-10-16' AND '2021-10-31'
),
user_product_stats AS (
    SELECT
        user_id,
        product_name,
        ROUND(
            (SUM(IF(status = 'no_completed', 1, 0)) / COUNT(*) * 100,
            2
        ) AS incomp_rate
    FROM filtered_orders
    GROUP BY user_id, product_name
    HAVING incomp_rate > 0
),
ranked_data AS (
    SELECT
        user_id,
        product_name,
        incomp_rate,
        DENSE_RANK() OVER (
            PARTITION BY product_name 
            ORDER BY incomp_rate DESC
        ) AS rnk
    FROM user_product_stats
)
SELECT
    product_name,
    user_id,
    rnk,
    CONCAT(incomp_rate, '%') AS incomp_rate
FROM ranked_data
WHERE rnk <= 3
ORDER BY product_name ASC, rnk ASC;
```
