#### App/core/base_algorithm. Py
```python
from abc import ABC, abstractmethod
from typing import Dict, Any, List
from dataclasses import dataclass
import numpy as np

@dataclass
class AlgorithmResult:
    """算法处理结果统一格式"""
    success: bool
    data: Any
    metadata: Dict[str, Any]
    error_message: str = None

class BaseAlgorithm(ABC):
    """算法基类 - 所有算法必须实现这个接口"""
    
    @abstractmethod
    def get_name(self) -> str:
        """返回算法唯一标识"""
        pass
    
    @abstractmethod
    def get_version(self) -> str:
        """返回算法版本"""
        pass
    
    @abstractmethod
    def get_description(self) -> str:
        """返回算法描述"""
        pass
    
    @abstractmethod
    def process_tile(self, tile: np.ndarray, **kwargs) -> AlgorithmResult:
        """处理单个瓦片 - 核心算法逻辑"""
        pass
    
    @abstractmethod
    def get_required_params(self) -> List[str]:
        """返回算法需要的参数列表"""
        pass
    
    def get_optional_params(self) -> Dict[str, Any]:
        """返回算法可选参数及默认值"""
        return {}
    
    def validate_input(self, tile: np.ndarray) -> bool:
        """验证输入数据格式（可选重写）"""
        return isinstance(tile, np.ndarray) and len(tile.shape) in [2, 3]
    
    def merge_results(self, results: List[AlgorithmResult]) -> AlgorithmResult:
        """合并多个瓦片的结果（可选重写）"""
        # 默认实现 - 简单合并
        successful_results = [r for r in results if r.success]
        if not successful_results:
            return AlgorithmResult(
                success=False,
                data=None,
                metadata={},
                error_message="All tile processing failed"
            )
        
        merged_data = [r.data for r in successful_results]
        merged_metadata = {
            "total_tiles": len(results),
            "successful_tiles": len(successful_results),
            "failed_tiles": len(results) - len(successful_results)
        }
        
        return AlgorithmResult(
            success=True,
            data=merged_data,
            metadata=merged_metadata
        )
```

#### App/core/base_algorithm. Py
```python
from abc import ABC, abstractmethod
from typing import Dict, Any, List
from dataclasses import dataclass
import numpy as np

@dataclass
class AlgorithmResult:
    """算法处理结果统一格式"""
    success: bool
    data: Any
    metadata: Dict[str, Any]
    error_message: str = None

class BaseAlgorithm(ABC):
    """算法基类 - 所有算法必须实现这个接口"""
    
    @abstractmethod
    def get_name(self) -> str:
        """返回算法唯一标识"""
        pass
    
    @abstractmethod
    def get_version(self) -> str:
        """返回算法版本"""
        pass
    
    @abstractmethod
    def get_description(self) -> str:
        """返回算法描述"""
        pass
    
    @abstractmethod
    def process_tile(self, tile: np.ndarray, **kwargs) -> AlgorithmResult:
        """处理单个瓦片 - 核心算法逻辑"""
        pass
    
    @abstractmethod
    def get_required_params(self) -> List[str]:
        """返回算法需要的参数列表"""
        pass
    
    def get_optional_params(self) -> Dict[str, Any]:
        """返回算法可选参数及默认值"""
        return {}
    
    def validate_input(self, tile: np.ndarray) -> bool:
        """验证输入数据格式（可选重写）"""
        return isinstance(tile, np.ndarray) and len(tile.shape) in [2, 3]
    
    def merge_results(self, results: List[AlgorithmResult]) -> AlgorithmResult:
        """合并多个瓦片的结果（可选重写）"""
        # 默认实现 - 简单合并
        successful_results = [r for r in results if r.success]
        if not successful_results:
            return AlgorithmResult(
                success=False,
                data=None,
                metadata={},
                error_message="All tile processing failed"
            )
        
        merged_data = [r.data for r in successful_results]
        merged_metadata = {
            "total_tiles": len(results),
            "successful_tiles": len(successful_results),
            "failed_tiles": len(results) - len(successful_results)
        }
        
        return AlgorithmResult(
            success=True,
            data=merged_data,
            metadata=merged_metadata
        )
```
